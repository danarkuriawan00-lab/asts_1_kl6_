Kode Google Apps Script (Copy & Paste) 
function doPost(e) {
  try {
    var sheet = SpreadsheetApp.getActiveSpreadsheet().getActiveSheet();
    var data = JSON.parse(e.postData.contents);
    sheet.appendRow([
      data.waktuSelesai,
      data.nama,
      data.kelas,
      data.noAbsen,
      data.tanggalLahir,
      data.mapel,
      data.nilai,
      data.ringkasan
    ]);
    return ContentService.createTextOutput("Success").setMimeType(ContentService.MimeType.TEXT);
  } catch (err) {
    return ContentService.createTextOutput("Error: " + err.toString()).setMimeType(ContentService.MimeType.TEXT);
  }
}

*PENTING: Saat Deploy pilih Execute as: Me (Saya) dan Who has access: Anyone (Siapa saja). Setiap ada perubahan kode di Apps Script, wajib lakukan New Deployment (Penerapan Baru).
